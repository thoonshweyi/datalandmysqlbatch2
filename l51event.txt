=>Event

Event Type (Schedule)
(i) One Time Event
(ii) Recurring Event

=>To see event was ON/OFF
SHOW PROCESSLIST;


SET GLOBAL event_scheduler = ON;
SET GLOBAL event_scheduler = OFF;

=>Syntax
Method 1

CREATE EVENT eventname
ON SCHEDULT AT timestamp
DO
statements
-------------------------
Method 2

CREATE EVENT eventname
ON SCHEDULT AT timestamp
ON COMPLETION PRESERVE
DO
statements

*can trace event history

(ii) Recurring Event

Method 1

CREATE EVENT eventname
ON SCHEDULT AT EVERY INTERVAL
STARTS timestamp
DO
statements
-------------------------

Method 2

CREATE EVENT eventname
ON SCHEDULT AT EVERY INTERVAL
STARTS timestamp
ENDS timestamp
DO
statements

CREATE TABLE IF NOT EXISTS vipcustomers(
     id INT NOT NULL AUTO_INCREMENT PRIMARY KEY,
     name VARCHAR(50) NOT NULL,
     email VARCHAR(50) NOT NULL UNIQUE,
     emailverify INT
);
DESC vipcustomers;

VALUES ("hla hla","hlahla@gmail.com"),
          ("thaw thaw","thawthaw@gmail.com");

SELECT * FROM vipcustomers;

CREATE EVENT vipcustomers_ev
ON SCHEDULE AT CURRENT_TIMESTAMP + INTERVAL 30 SECOND
DO
DELETE FROM vipcustomers WHERE emailverify IS NULL;
